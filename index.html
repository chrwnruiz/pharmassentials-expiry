function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("MAIN SHEET");
  var lastRow = sheet.getLastRow();
  var items = [];

  if (lastRow >= 4) {
    var values = sheet.getRange(4, 9, lastRow - 3, 3).getValues();
    for (var i = 0; i < values.length; i++) {
      var category = values[i][0] ? values[i][0].toString().trim() : "Others";
      var name = values[i][1] ? values[i][1].toString().trim() : "";
      var expiryRaw = values[i][2];

      if (name !== "") {
        var expiryStr = "";
        if (expiryRaw instanceof Date) {
          expiryStr = Utilities.formatDate(expiryRaw, Session.getScriptTimeZone(), "MMM yyyy");
        } else if (expiryRaw) {
          expiryStr = expiryRaw.toString().trim();
        }
        items.push({
          row: i + 4,
          category: category || "Others",
          name: name,
          expiry: expiryStr
        });
      }
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok", items: items }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("MAIN SHEET");
    var action = payload.action;

    if (action === "updateExpiry") {
      var targetName = (payload.name || "").toString().trim().toUpperCase();
      var newExpiry = (payload.expiry || "").toString().trim();
      var rowNum = parseInt(payload.row, 10);
      var updated = false;

      if (rowNum >= 4) {
        var rowCellName = sheet.getRange(rowNum, 10).getValue().toString().trim().toUpperCase();
        if (rowCellName === targetName) {
          sheet.getRange(rowNum, 11).setValue(newExpiry);
          updated = true;
        }
      }

      if (!updated) {
        var lastRow = sheet.getLastRow();
        if (lastRow >= 4) {
          var names = sheet.getRange(4, 10, lastRow - 3, 1).getValues();
          for (var i = 0; i < names.length; i++) {
            if (names[i][0].toString().trim().toUpperCase() === targetName) {
              sheet.getRange(i + 4, 11).setValue(newExpiry);
              updated = true;
              break;
            }
          }
        }
      }

      return ContentService
        .createTextOutput(JSON.stringify({ status: "ok", updated: updated }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === "addProduct") {
      var cat = (payload.category || "Others").toString().trim();
      var prodName = (payload.name || "").toString().trim();
      var prodExpiry = (payload.expiry || "").toString().trim();

      if (!prodName) {
        throw new Error("Product name is required.");
      }

      var lastRowAll = sheet.getLastRow();
      var insertRow = 4;
      if (lastRowAll >= 4) {
        var colJ = sheet.getRange(4, 10, lastRowAll - 3, 1).getValues();
        for (var r = colJ.length - 1; r >= 0; r--) {
          if (colJ[r][0].toString().trim() !== "") {
            insertRow = r + 5;
            break;
          }
        }
      }

      sheet.getRange(insertRow, 9, 1, 3).setValues([[cat, prodName, prodExpiry]]);

      return ContentService
        .createTextOutput(JSON.stringify({ status: "ok", row: insertRow }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: "Unknown action" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
